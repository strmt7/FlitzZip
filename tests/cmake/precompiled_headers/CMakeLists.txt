cmake_minimum_required(VERSION 3.24)
project(SuperZipPrecompiledHeaderContract LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT IS_DIRECTORY "${REPO_ROOT}/src")
  message(FATAL_ERROR "REPO_ROOT must identify the actual repository")
endif()
foreach(target IN ITEMS superzip_core superzip_test_objects SuperZip decoy)
  add_executable("${target}" contract.cpp contract.c)
  target_compile_definitions("${target}" PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endforeach()
target_compile_definitions(SuperZip PRIVATE UNICODE _UNICODE
                                            EXPECTED_CHARACTER_BYTES=2)
foreach(target IN ITEMS superzip_core superzip_test_objects decoy)
  target_compile_definitions("${target}" PRIVATE EXPECTED_CHARACTER_BYTES=1)
endforeach()
include("${REPO_ROOT}/cmake/SuperZipPrecompiledHeaders.cmake")
superzip_enable_precompiled_headers()
foreach(target IN ITEMS superzip_core superzip_test_objects SuperZip)
  get_target_property(headers "${target}" PRECOMPILE_HEADERS)
  get_target_property(inherited "${target}" INTERFACE_PRECOMPILE_HEADERS)
  get_target_property(reused "${target}" PRECOMPILE_HEADERS_REUSE_FROM)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
    list(LENGTH headers _header_count)
    if(NOT _header_count EQUAL 18
       OR inherited
       OR reused)
      message(FATAL_ERROR "PCH must be private and independently compiled")
    endif()
    foreach(header IN LISTS headers)
      if(NOT header MATCHES
         "^\\$<\\$<COMPILE_LANGUAGE:CXX>:<[a-z_]+\\$<ANGLE-R>>$")
        message(FATAL_ERROR "PCH must contain only C++ standard headers")
      endif()
    endforeach()
  elseif(headers)
    message(FATAL_ERROR "This optimization has only been qualified for MSVC")
  endif()
endforeach()
get_target_property(decoy_headers decoy PRECOMPILE_HEADERS)
if(decoy_headers)
  message(FATAL_ERROR "PCH escaped its explicit target cohort")
endif()
